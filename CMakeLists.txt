cmake_minimum_required(VERSION 3.31.6)
# Compiler metadata does not require a disposable Windows executable.
if(CMAKE_HOST_WIN32 AND NOT CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
endif()
project(MEU VERSION 0.1.1 LANGUAGES C)
if(CMAKE_HOST_WIN32 AND NOT CMAKE_TOOLCHAIN_FILE)
  set(CMAKE_TRY_COMPILE_TARGET_TYPE EXECUTABLE)
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(MEU_BUILD_CLIENT "Build the SDL/OpenGL client" ON)
option(MEU_DIAGNOSTICS "Expose read-only browser diagnostics for automated tests" OFF)
option(MEU_SANITIZERS "Enable address/undefined sanitizers for native core tests" OFF)
include(CTest)
if(NOT MEU_REVISION)
  if(DEFINED ENV{GITHUB_SHA} AND "$ENV{GITHUB_SHA}" MATCHES "^[0-9a-fA-F]+$")
    # Container checkouts may not be visible to git under the build user's HOME.
    set(MEU_REVISION "$ENV{GITHUB_SHA}")
  else()
    execute_process(COMMAND git describe --always --dirty WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
      OUTPUT_VARIABLE MEU_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  endif()
endif()
if(NOT MEU_REVISION)
  set(MEU_REVISION "source")
endif()
configure_file(src/core/build_info.h.in generated/meu_build_info.h @ONLY)
add_library(meu_core STATIC src/game/expedition.c src/game/power.c src/core/simulation.c src/core/settings.c src/core/snapshot.c src/core/matrix.c src/core/pose.c src/game/interior.c src/game/travel.c src/world/travel_world.c)
target_include_directories(meu_core PUBLIC src "${CMAKE_BINARY_DIR}/generated")
if(NOT MSVC)
  target_compile_options(meu_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(NOT WIN32)
    target_link_libraries(meu_core PUBLIC m)
  endif()
endif()
if(MEU_SANITIZERS AND NOT EMSCRIPTEN AND NOT MSVC)
  target_compile_options(meu_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(meu_core PUBLIC -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
  add_executable(meu_mesh_tests tests/unit/mesh_tests.c src/render/mesh.c)
  target_link_libraries(meu_mesh_tests PRIVATE meu_core)
  add_executable(meu_expedition_tests tests/unit/expedition_tests.c)
  target_link_libraries(meu_expedition_tests PRIVATE meu_core)
  add_executable(meu_power_tests tests/unit/power_tests.c)
  target_link_libraries(meu_power_tests PRIVATE meu_core)
  add_executable(meu_snapshot_tests tests/unit/snapshot_tests.c)
  target_link_libraries(meu_snapshot_tests PRIVATE meu_core)
  if(NOT EMSCRIPTEN)
    add_executable(meu_file_storage_tests tests/unit/file_storage_tests.c src/platform/file_storage.c)
    target_compile_definitions(meu_file_storage_tests PRIVATE MEU_STORAGE_TESTING=1)
    target_link_libraries(meu_file_storage_tests PRIVATE meu_core)
    add_test(NAME file_storage COMMAND meu_file_storage_tests "${CMAKE_BINARY_DIR}/test-storage")
  endif()
  add_executable(meu_core_tests tests/unit/core_tests.c)
  target_link_libraries(meu_core_tests PRIVATE meu_core)
  add_executable(meu_interior_tests tests/unit/interior_tests.c)
  target_link_libraries(meu_interior_tests PRIVATE meu_core)
  add_executable(meu_travel_tests tests/unit/travel_tests.c)
  target_link_libraries(meu_travel_tests PRIVATE meu_core)
  add_executable(meu_settings_tests tests/unit/settings_tests.c)
  target_link_libraries(meu_settings_tests PRIVATE meu_core)
  add_executable(meu_scenarios tests/scenarios/scenario_runner.c)
  target_link_libraries(meu_scenarios PRIVATE meu_core)
  if(EMSCRIPTEN)
    find_program(MEU_NODE node REQUIRED)
    foreach(test_target meu_mesh_tests meu_expedition_tests meu_power_tests meu_snapshot_tests meu_core_tests meu_scenarios meu_interior_tests meu_travel_tests meu_settings_tests)
      set_target_properties(${test_target} PROPERTIES SUFFIX ".js")
      target_link_options(${test_target} PRIVATE "-sENVIRONMENT=node" "-sEXIT_RUNTIME=1" "-sNODERAWFS=1")
    endforeach()
    add_test(NAME meshes COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_mesh_tests>")
    add_test(NAME expedition COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_expedition_tests>")
    add_test(NAME power COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_power_tests>")
    add_test(NAME snapshot_scenario COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_scenarios>" "${PROJECT_SOURCE_DIR}/tests/fixtures/m0e-snapshot.txt")
    add_test(NAME snapshots COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_snapshot_tests>" "${PROJECT_SOURCE_DIR}/tests/fixtures/save-v2.meu")
    add_test(NAME core COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_core_tests>")
    add_test(NAME settings COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_settings_tests>")
    add_test(NAME settings_scenario COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_scenarios>" "${PROJECT_SOURCE_DIR}/tests/fixtures/m0d-settings.txt")
    add_test(NAME travel COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_travel_tests>")
    add_test(NAME interior COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_interior_tests>")
    add_test(NAME interior_tour COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_scenarios>" "${PROJECT_SOURCE_DIR}/tests/fixtures/m0b-tour.txt")
    add_test(NAME piloting_scenario COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_scenarios>" "${PROJECT_SOURCE_DIR}/tests/fixtures/m0c-piloting.txt")
    add_test(NAME travel_collision_scenario COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_scenarios>" "${PROJECT_SOURCE_DIR}/tests/fixtures/m0c-collision.txt")
    add_test(NAME walk_scenario COMMAND "${MEU_NODE}" "$<TARGET_FILE:meu_scenarios>"
      "${PROJECT_SOURCE_DIR}/tests/fixtures/m0a-walk.txt")
  else()
    add_test(NAME meshes COMMAND meu_mesh_tests)
    add_test(NAME expedition COMMAND meu_expedition_tests)
    add_test(NAME power COMMAND meu_power_tests)
    add_test(NAME snapshot_scenario COMMAND meu_scenarios "${PROJECT_SOURCE_DIR}/tests/fixtures/m0e-snapshot.txt")
    add_test(NAME snapshots COMMAND meu_snapshot_tests "${PROJECT_SOURCE_DIR}/tests/fixtures/save-v2.meu")
    add_test(NAME core COMMAND meu_core_tests)
    add_test(NAME settings COMMAND meu_settings_tests)
    add_test(NAME settings_scenario COMMAND meu_scenarios "${PROJECT_SOURCE_DIR}/tests/fixtures/m0d-settings.txt")
    add_test(NAME travel COMMAND meu_travel_tests)
    add_test(NAME interior COMMAND meu_interior_tests)
    add_test(NAME interior_tour COMMAND meu_scenarios "${PROJECT_SOURCE_DIR}/tests/fixtures/m0b-tour.txt")
    add_test(NAME piloting_scenario COMMAND meu_scenarios "${PROJECT_SOURCE_DIR}/tests/fixtures/m0c-piloting.txt")
    add_test(NAME travel_collision_scenario COMMAND meu_scenarios "${PROJECT_SOURCE_DIR}/tests/fixtures/m0c-collision.txt")
    add_test(NAME walk_scenario COMMAND meu_scenarios "${PROJECT_SOURCE_DIR}/tests/fixtures/m0a-walk.txt")
  endif()
endif()
if(MEU_BUILD_CLIENT)
  if(NOT EMSCRIPTEN)
    add_library(meu_file_storage STATIC src/platform/file_storage.c)
    target_link_libraries(meu_file_storage PUBLIC meu_core)
    if(NOT MSVC)
      target_compile_options(meu_file_storage PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
  endif()
  include(FetchContent)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL3
    URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.18/SDL3-3.4.18.tar.gz
    URL_HASH SHA256=9c75cf16330322c217dedd2e0609f1124f1b54b8633e763467b4684d0f4334a3
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL3)
  # SDL's Emscripten backend uses GNU EM_ASM macros; MEU's own targets stay strict C17.
  set_target_properties(SDL3-static PROPERTIES C_EXTENSIONS ON)
  add_executable(meu src/platform/main.c src/render/renderer.c src/render/mesh.c)
  target_link_libraries(meu PRIVATE meu_core SDL3::SDL3)
  if(NOT MSVC)
    target_compile_options(meu PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(MEU_DIAGNOSTICS)
    target_compile_definitions(meu PRIVATE MEU_DIAGNOSTICS=1)
  endif()
  if(EMSCRIPTEN)
    set_target_properties(meu PROPERTIES OUTPUT_NAME "index" SUFFIX ".html")
    set(MEU_EXPORTS "_main,_meu_web_station,_meu_web_station_text,_meu_web_action,_meu_web_look,_meu_web_start,_meu_web_pause,_meu_web_settings,_meu_web_setting,_meu_web_journeys,_meu_web_snapshot,_meu_web_probe,_meu_web_restore,_meu_web_save_message")
    if(MEU_DIAGNOSTICS)
      string(APPEND MEU_EXPORTS ",_meu_state_json")
    endif()
    target_link_options(meu PRIVATE
      "--shell-file=${PROJECT_SOURCE_DIR}/web/shell.html"
      "-sMIN_WEBGL_VERSION=2" "-sMAX_WEBGL_VERSION=2"
      "-sALLOW_MEMORY_GROWTH=1" "-sINITIAL_MEMORY=33554432" "-sMAXIMUM_MEMORY=268435456"
      "-sENVIRONMENT=web" "-sEXPORTED_FUNCTIONS=${MEU_EXPORTS}" "-sEXPORTED_RUNTIME_METHODS=ccall"
      "-sASSERTIONS=$<IF:$<CONFIG:Release>,0,1>" "-sEXIT_RUNTIME=1")
    configure_file(web/storage.js "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/storage.js" COPYONLY)
    configure_file(web/input.js "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/input.js" COPYONLY)
    configure_file(web/style.css "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/style.css" COPYONLY)
  else()
    find_package(OpenGL REQUIRED)
    target_link_libraries(meu PRIVATE OpenGL::GL meu_file_storage)
    if(WIN32 AND MINGW)
      target_link_options(meu PRIVATE -static-libgcc)
    endif()
  endif()
endif()
